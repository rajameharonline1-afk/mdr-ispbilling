# nano /etc/apache2/sites-available/isp_billing.conf

<VirtualHost *:80>
    ServerAdmin webmaster@localhost

    # 1. DocumentRoot শুধু public ফোল্ডারে সেট করা থাকবে
    DocumentRoot /var/www/isp_billing/public
    ServerName 172.16.3.3

    # 2. শুধু public ডিরেক্টরির জন্য অ্যাক্সেস অনুমোদিত
    <Directory /var/www/isp_billing/public>
        DirectoryIndex index.php index.html
        Options FollowSymLinks
        AllowOverride All
        Require all granted
    </Directory>

    # --- প্রজেক্টের রুটে থাকা বাকি ফোল্ডার যেগুলো HTTP দিয়ে এক্সেস লাগে ---
    Alias /app     /var/www/isp_billing/app
    Alias /api     /var/www/isp_billing/api
    Alias /cron    /var/www/isp_billing/cron
    Alias /reports /var/www/isp_billing/reports
    Alias /tools   /var/www/isp_billing/tools
    Alias /tg      /var/www/isp_billing/tg

    <Directory /var/www/isp_billing/app>
        AllowOverride All
        Require all granted
    </Directory>
    <Directory /var/www/isp_billing/api>
        AllowOverride All
        Require all granted
    </Directory>
    <Directory /var/www/isp_billing/cron>
        AllowOverride All
        Require all granted
    </Directory>
    <Directory /var/www/isp_billing/reports>
        AllowOverride All
        Require all granted
    </Directory>
    <Directory /var/www/isp_billing/tools>
        AllowOverride All
        Require all granted
    </Directory>
    <Directory /var/www/isp_billing/tg>
        AllowOverride All
        Require all granted
    </Directory>

    # 3. পুরোনো বুকমার্কে যদি /public/... থাকে তাহলে সেই পথকে নতুন root-এ নিয়ে আসবো
    RedirectMatch permanent "^/public/(.*)$" "/$1"

    # 4. রুট URL কে login.php-তে রিডাইরেক্ট করবে (এই লাইনটি রাখুন)
    RedirectMatch permanent "^/$" "/login.php"

    ErrorLog ${APACHE_LOG_DIR}/isp_billing-error.log
    CustomLog ${APACHE_LOG_DIR}/isp_billing-access.log combined
</VirtualHost>
